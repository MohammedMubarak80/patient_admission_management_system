conn.commit()

# ==========================================
# ADD PATIENT API
# ==========================================

@app.route("/api/patients", methods=["POST"])
def add_patient():

    data = request.get_json(silent=True) or {}

    required = [
        "name", "age", "gender", "ward",
        "department", "doctor", "admission_date"
    ]

    if any(not str(data.get(k, "")).strip() for k in required):
        return jsonify({"error": "All fields are required"}), 400

    if data["gender"] not in GENDERS or data["ward"] not in WARDS:
        return jsonify({"error": "Invalid gender or ward"}), 400

    if data["department"] not in DEPARTMENTS or data["doctor"] not in DOCTORS:
        return jsonify({"error": "Invalid department or doctor"}), 400

    try:
        age = int(data["age"])
        if age < 0 or age > 120:
            return jsonify({"error": "Invalid age"}), 400
    except (ValueError, TypeError):
        return jsonify({"error": "Age must be a number"}), 400

    conn = get_db()

    cursor = conn.execute("""
    INSERT INTO patients
    (name, age, gender, ward, department,
     doctor, admission_date, status)
    VALUES (?, ?, ?, ?, ?, ?, ?, 'Admitted')
    """, (
        data["name"].strip(), age, data["gender"],
        data["ward"], data["department"],
        data["doctor"], data["admission_date"]
    ))

    patient_id = cursor.lastrowid

    conn.commit()
    conn.close()

    return jsonify({
        "message": "Patient added successfully!",
        "patient_id": patient_id
    }), 201


# ==========================================
# GET PATIENTS API
# ==========================================

@app.route("/api/patients", methods=["GET"])
def get_patients():

    conn = get_db()

    rows = conn.execute(
        "SELECT * FROM patients ORDER BY id DESC"
    ).fetchall()

    conn.close()

    return jsonify([dict(row) for row in rows])


# ==========================================
# UPDATE PATIENT API
# ==========================================

@app.route("/api/patients/<int:patient_id>",
           methods=["PUT"])
def update_patient(patient_id):

    data = request.get_json(silent=True) or {}

    allowed = [
        "name", "age", "gender", "ward",
        "department", "doctor", "admission_date"
    ]

    updates = {
        key: data[key]
        for key in allowed if key in data
    }

    if not updates:
        return jsonify({"error": "No fields to update"}), 400

    if "name" in updates and not str(updates["name"]).strip():
        return jsonify({"error": "Name cannot be empty"}), 400

    if "age" in updates:
        try:
            updates["age"] = int(updates["age"])
            if updates["age"] < 0 or updates["age" ] > 120:
                raise ValueError()
        except (ValueError, TypeError):
            return jsonify({"error": "Invalid age"}), 400

    for key, allowed_values in [
        ("gender", GENDERS),
        ("ward", WARDS),
        ("department", DEPARTMENTS),
        ("doctor", DOCTORS)
    ]:
        if key in updates and updates[key] not in allowed_values:
            return jsonify({"error": f"Invalid {key}"}), 400

    conn = get_db()

    exists = conn.execute(
        "SELECT id FROM patients WHERE id=?",
        (patient_id,)
    ).fetchone()

    if not exists:
        conn.close;
        return jsonify({"error": "Patient not found"}), 404

    query = ", ".join(f"{key}=?" for key in updates)

    conn.execute(
        f"UPDATE patients SET {query} WHERE id=?",
        list(updates.values()) + [patient_id]
    )

    conn.commit()
    conn.close()

    return jsonify({
        "message": "Patient updated successfully!"
    })


# ==========================================
# DISCHARGE API
# ==========================================

@app.route("/api/patients/<int:patient_id>/discharge",
           methods=["PATCH"])
def discharge_patient(patient_id):

    conn = get_db()

    patient = conn.execute(
        "SELECT id, status FROM patients WHERE id=?",
        (patient_id,)
    ).fetchone()

    if not patient:
        conn.close()
        return jsonify({"error": "Patient not found"}), 404

    if patient["status"] == "Discharged":
        conn.close()
        return jsonify({"message": "Patient is already discharged"})

    conn.execute(
        "UPDATE patients SET status='Discharged' WHERE id=?",
        (patient_id,)
    )

    conn.commit()
    conn.close()

    return jsonify({
        "message": "Patient discharged successfully!"
    })
